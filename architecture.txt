% Appendix H. CNN used to remove whole frame artefact images

imds = imageDatastore('C:\Users\wendi\Documents\MATLAB\Noise_DNA\');
goundtruth = readtable("Noise_DNAlabels.csv",'Delimiter', ',', 'HeaderLines', 0,...
'ReadVariableNames', true);
imds.Labels =  categorical(goundtruth.Status);

% calculate the number of images in each category
labelCount = countEachLabel(imds)

% in order to specify the size of the images in the input layer, check the
% size of the first image
img = readimage(imds,1);
size(img)

% divide the data into training and testing sets where each category
% contains 750 images
numTrainFiles = 900;
[imdsTrain,imdsTest] = splitEachLabel(imds,numTrainFiles,'randomize');

% define cnn architecture
layers = [
   % using image size from line 21 and color channel for grayscale
   imageInputLayer([250 250 1])
  
   % filter size 3x3 with 8 neurons(filters) where spatial output size is
   % the same as the input size
    convolution2dLayer(7,8,'Padding','same')
   % normalize the activations and gradients to speed up network training
    % and reduce sensitivity to network initialization
    batchNormalizationLayer
    % nonlinear activation function (rectified linear unit)
    reluLayer
   
   % down sampling operation to reduce spatial size of the feature map and
    % remove redundant spatial information. this layer will return the max
    % values of rectangular regions of inputs, specified by the first
    % argument followed by the 'Stride' name-value pair to specify the step%     % size
    maxPooling2dLayer(2,'Stride',2)
   
     % increase in neurons
     convolution2dLayer(7,16,'Padding','same')
     batchNormalizationLayer
     reluLayer
    
     maxPooling2dLayer(2,'Stride',2)
   
    % increase in neurons
    convolution2dLayer(7,32,'Padding','same')
    batchNormalizationLayer
    reluLayer
   
    maxPooling2dLayer(2,'Stride',2)
   
   convolution2dLayer(7,64,'Padding','same')
   batchNormalizationLayer
   reluLayer
  
   maxPooling2dLayer(2,'Stride',2)
  
   convolution2dLayer(7,128,'Padding','same')
   batchNormalizationLayer
   reluLayer
  
  
   % this layer is where the neurons connect to all the neurons in the
   % proceeding layer. combines all the features learned by the
   % previous layers to identify larger patterns. combines features to
   % classify the images therefore the OutputSize is equal to the number
   % of classes
   fullyConnectedLayer(2)

   % normalize the output of the fully connected layer
   softmaxLayer

   %  this layer uses the probabilities returned by the softmax function
   %  for each input to assign the input to one of the mutually exclusive
   %  classes and compute the loss
   classificationLayer];

% specify training options
% use sgdm with InitialLearnRate 0.01, set max number of epochs to 4,
% shuffle the data every epoch, monitor network accuracy during training
% by specifying validation data and frequency. validation data is
% not used to update the network weights. turn on the training progress
% plot, and turn off the command window output
options = trainingOptions('sgdm','InitialLearnRate',0.001,'MaxEpochs',30,...
   'Shuffle','every-epoch','ValidationData',imdsTest,'ValidationFrequency',...
   30,'Verbose',false,'Plots','training-progress');

% train the network
net = trainNetwork(imdsTrain,layers,options);
% save the trained network
DNALrgFiltnet = net;
save DNALrgFiltnet

% classify images and compute accuracy
YPred = classify(net,imdsTest);
YValidation = imdsTest.Labels;
accuracy = sum(YPred == YValidation)/numel(YValidation)
figure;
confusionchart(YValidation,YPred)

% Appendix J.  Script to sharpen edges using the Prewitt method

function [] = dataPrep(P,outpath,path,framefile)

I = imread([path,framefile]);

% convert the pixel values into nanometers
[I_nm] = pix2nm(I,framefile,P);

% threshold
IT = I_nm > P.tcutoff;

% remove objects from the border
I_clear = imclearborder(IT,8);

% fill in objects
% SE = strel('disk',2,8);

I_obj = I_nm;
IT_not = ~I_clear;    % inverse of IT
I_obj(IT_not) = 0;   % make all pixels that aren't part of objects = 0

% isolate the remaining objects left in the image
cc = bwconncomp(I_clear,8);
particleAreaS = regionprops(I_clear, 'Area');
particleArea = [particleAreaS.Area];
boolParticleKeep = particleArea > P.areacutoff;

if isempty(particleArea)
   return
end

pxIdx = cc.PixelIdxList;
pxIdxFilt = pxIdx(boolParticleKeep);
if isempty(pxIdxFilt)
   return
end

[threshIM] = thresh(pxIdxFilt,I,I_obj);
% applying edges over image
edgethresh = edge(threshIM,'Prewitt',.049);

threshmask = imoverlay(threshIM,edgethresh,'white');
graymask = rgb2gray(threshmask);
threshfinal = im2uint16(graymask);

% assigning pixel values to the false background
falsethresh = threshfinal == false;
threshfinal(falsethresh) = 16000;
dest = sprintf('%s_%s.tiff',...
outpath,framefile);
imwrite(threshfinal,dest)
% imwrite(threshIM,dest)
end
